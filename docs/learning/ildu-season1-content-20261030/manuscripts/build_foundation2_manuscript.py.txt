def jdump(obj):
    return json.dumps(obj, ensure_ascii=False, separators=(",", ":"))

def cell(value):
    if not isinstance(value, str):
        value = jdump(value)
    return value.replace("|", "\\|").replace("\n", "<br>")

def table(headers, rows):
    return "\n".join(["| " + " | ".join(headers) + " |", "|" + "---|" * len(headers)] + ["| " + " | ".join(cell(v) for v in row) + " |" for row in rows])

def triad(rows):
    return table(["구분 / 화자", "한국어", "English", "Deutsch (de-DE)"], rows)

INITIAL = "ㄱㄲㄴㄷㄸㄹㅁㅂㅃㅅㅆㅇㅈㅉㅊㅋㅌㅍㅎ"
VOWEL = "ㅏㅐㅑㅒㅓㅔㅕㅖㅗㅘㅙㅚㅛㅜㅝㅞㅟㅠㅡㅢㅣ"

def pieces(s):
    off = ord(s) - 0xAC00
    assert 0 <= off < 11172 and off % 28 == 0, s
    return [INITIAL[off // 588], VOWEL[(off % 588) // 28]]

def shape(s):
    ini, vow = pieces(s)
    return f"{ini} 위 / {vow} 아래" if vow in "ㅗㅜㅡ" else f"{ini} 왼쪽 / {vow} 오른쪽"

def add(l, stage, num, kind, stimulus, opts, answer, prompt, scope, feedback):
    rec = {"unit_id": l["id"], "item_id": f"B{l['n']:02d}-{stage}{num:02d}", "stage": stage,
           "kind": kind, "stimulus": stimulus, "options_author_only": opts,
           "answer_author_only": answer, "prompt": prompts[prompt], "scope": scope,
           "feedback": feedback}
    items.append(rec)
    for field in ("audio",):
        if field in stimulus:
            audios.add(stimulus[field])
    if kind == "read_audio":
        audios.update(opts)
    return rec

def audio_opts(target, initial_distractor, vowel_distractor, rotation):
    choices = [target, initial_distractor, vowel_distractor]
    return choices[rotation:] + choices[:rotation]
