def parts(s):
 n=ord(s)-44032
 assert 0<=n<11172,s
 return [INITIAL[n//588],VOWEL[(n%588)//28]]+([FINAL[n%28]] if n%28 else [])

def copula(s):
 return '이에요' if (ord(s[-1])-44032)%28 else '예요'

def shape(s):
 p=parts(s)
 if p[1] in 'ㅘㅝㅟㅚ': base=f'{p[0]} 왼쪽 위 / {p[1]} 오른쪽과 아래'
 elif p[1] in 'ㅗㅛㅜㅠㅡ': base=f'{p[0]} 위 / {p[1]} 그 아래'
 else: base=f'{p[0]} 왼쪽 / {p[1]} 오른쪽'
 return base+(f' / 받침 {p[2]} 가장 아래' if len(p)==3 else '')

def jd(o): return json.dumps(o,ensure_ascii=False,separators=(',',':'))
def cell(o): return (o if isinstance(o,str) else jd(o)).replace('|','\\|').replace('\n','<br>')
def table(headers,rows): return '\n'.join(['| '+' | '.join(headers)+' |','|'+'---|'*len(headers)]+['| '+' | '.join(cell(v) for v in row)+' |' for row in rows])
def tri(rows): return table(['구분 / 화자','한국어','English','Deutsch (de-DE)'],rows)

def rotated(options,n):
 n=n%len(options)
 return options[n:]+options[:n]

def detail(s):

 de=f'Anfangszeichen {p[0]}, Vokal {p[1]}'+(f', Endkonsonant {p[2]}' if len(p)==3 else '')
 return ko,en,de
