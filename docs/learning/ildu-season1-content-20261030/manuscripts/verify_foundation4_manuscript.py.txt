INI='ㄱㄲㄴㄷㄸㄹㅁㅂㅃㅅㅆㅇㅈㅉㅊㅋㅌㅍㅎ'
VOW='ㅏㅐㅑㅒㅓㅔㅕㅖㅗㅘㅙㅚㅛㅜㅝㅞㅟㅠㅡㅢㅣ'
FIN='\0ㄱㄲㄳㄴㄵㄶㄷㄹㄺㄻㄼㄽㄾㄿㅀㅁㅂㅄㅅㅆㅇㅈㅊㅋㅌㅍㅎ'
def parts(s):
 n=ord(s)-44032
 return INI[n//588],VOW[(n%588)//28],FIN[n%28] if n%28 else ''
def check_text(value,known,ctx):
 if isinstance(value,list):
  for v in value:check_text(v,known,ctx)
 elif isinstance(value,str):
  for ch in value:
   if 44032<=ord(ch)<55204:
    i,v,f=parts(ch)
    assert i in known['i'] and v in known['v'] and (not f or f in known['f']),(ctx,value,ch,known)
   elif ch in VOW:assert ch in known['v'],(ctx,ch)
   elif ch in INI:assert ch in known['i'] or ch in known['f'],(ctx,ch)

assert len(items)==len(rows)==117 and len({r['item_id'] for r in items})==117

audio_needed=set();positions=Counter();by_id={r['item_id']:r for r in items}

def phon_key(text):
 text=linked.get(text,text)
 key=[]
 for ch in text:
  if 44032<=ord(ch)<55204:
   i,v,f=parts(ch)
   if f and f in 'ㅅㅆㅈㅊㅌ':f='ㄷ'
   if f and f in 'ㄲㅋ':f='ㄱ'
   if f=='ㅍ':f='ㅂ'
   key.append((i,v,f))
  else:key.append(ch)
 return repr(key)
